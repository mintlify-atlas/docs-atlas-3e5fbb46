title: React
description: Use CVA in React components with Tailwind CSS or CSS Modules to build type-safe, variant-driven UI.

CVA integrates naturally with React. You define your variants in a standalone function, then spread the result into your component's `className` prop. `VariantProps` gives you full TypeScript inference on the props your component accepts.

<Tabs>
  <Tab title="Tailwind CSS">
    Define your variant function alongside your component, then extend `React.ButtonHTMLAttributes` with `VariantProps` to get a fully-typed component interface.

    ```tsx components/button.tsx
    import React from "react";
    import { cva, type VariantProps } from "class-variance-authority";

    const button = cva(
      ["font-semibold", "border", "rounded", "transition-colors"],
      {
        variants: {
          intent: {
            primary: [
              "bg-blue-500",
              "text-white",
              "border-transparent",
            ],
            secondary: [
              "bg-white",
              "text-gray-800",
              "border-gray-400",
            ],
          },
          size: {
            small: ["text-sm", "py-1", "px-2"],
            medium: ["text-base", "py-2", "px-4"],
          },
          disabled: {
            true: ["opacity-50", "cursor-not-allowed"],
            false: null,
          },
        },
        compoundVariants: [
          {
            intent: "primary",
            disabled: false,
            class: "hover:bg-blue-600",
          },
          {
            intent: "secondary",
            disabled: false,
            class: "hover:bg-gray-100",
          },
          { intent: "primary", size: "medium", class: "uppercase" },
        ],
        defaultVariants: {
          intent: "primary",
          size: "medium",
          disabled: false,
        },
      }
    );

    export interface ButtonProps
      extends Omit<React.ButtonHTMLAttributes<HTMLButtonElement>, "disabled">,
        VariantProps<typeof button> {}

    export const Button: React.FC<ButtonProps> = ({
      className,
      intent,
      size,
      disabled,
      ...props
    }) => (
      <button
        className={button({ intent, size, disabled, className })}
        disabled={disabled || undefined}
        {...props}
      />
    );
    ```

    <Tip>
      Enable Tailwind CSS IntelliSense inside `cva` calls by adding `"cva"` to `tailwindCSS.classFunctions` in your editor settings. See the [Tailwind CSS guide](/guides/tailwind-css) for details.
    </Tip>
  </Tab>
  <Tab title="CSS Modules">
    Pass CSS Module class names directly as variant values. CVA treats them as plain strings — no adapter or plugin needed.

    ```tsx components/button.tsx
    import React from "react";
    import { cva, type VariantProps } from "class-variance-authority";
    import styles from "./button.module.css";

    const button = cva(styles.base, {
      variants: {
        intent: {
          primary: styles.primary,
          secondary: styles.secondary,
        },
        size: {
          small: styles.small,
          medium: styles.medium,
        },
        disabled: {
          false: styles.enabled,
          true: styles.disabled,
        },
      },
      compoundVariants: [
        { intent: "primary", size: "medium", className: styles.primaryMedium },
      ],
      defaultVariants: {
        intent: "primary",
        size: "medium",
        disabled: false,
      },
    });

    export interface ButtonProps
      extends Omit<React.ButtonHTMLAttributes<HTMLButtonElement>, "disabled">,
        VariantProps<typeof button> {}

    export const Button: React.FC<ButtonProps> = ({
      className,
      intent,
      size,
      disabled,
      ...props
    }) => (
      <button
        className={button({ intent, size, disabled, className })}
        disabled={disabled || undefined}
        {...props}
      />
    );
    ```

    See the [CSS Modules example](/examples/css-modules) for the matching stylesheet and a more detailed walkthrough.

## Splitting variants from the component

For larger components, you can define your variants in a separate file and import them where needed. This also makes it easy to share a variant function across multiple framework components.

```typescript components/button.variants.ts
import { cva, type VariantProps } from "class-variance-authority";

export const buttonVariants = cva(["font-semibold", "border", "rounded"], {
  variants: {
    intent: {
      primary: ["bg-blue-500", "text-white", "border-transparent"],
      secondary: ["bg-white", "text-gray-800", "border-gray-400"],
    },
    size: {
      small: ["text-sm", "py-1", "px-2"],
      medium: ["text-base", "py-2", "px-4"],
    },
  },
  defaultVariants: {
    intent: "primary",
    size: "medium",
  },
});

export type ButtonVariantProps = VariantProps<typeof buttonVariants>;
```

```tsx components/button.tsx
import React from "react";
import { buttonVariants, type ButtonVariantProps } from "./button.variants";

export interface ButtonProps
  extends React.ButtonHTMLAttributes<HTMLButtonElement>,
    ButtonVariantProps {}

export const Button: React.FC<ButtonProps> = ({
  className,
  intent,
  size,
  ...props
}) => (
  <button
    className={buttonVariants({ intent, size, className })}
    {...props}
  />
);
```
